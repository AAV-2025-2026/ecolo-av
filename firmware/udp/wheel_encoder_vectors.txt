# Checked by:
#   firmware/spine/crates/wheel-encoder/tests/vectors.rs   (the Pico 2's Rust)
#   ros/can_bridge/aav_legacy/test/test_legacy_protocol.py   (encoder_udp on the Motor Pi)
#
